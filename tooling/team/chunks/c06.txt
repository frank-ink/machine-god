def _ids(db_path: str, star_dir: str) -> dict:
    # same derivation idea as the donor: installation id hangs off DB path + code dir
    inst = hashlib.sha256(f"F2WP1207_INSTALLATION_REF/v1:{db_path}:{star_dir}".encode()).hexdigest()[:32]
    return {
        "entity_id": hashlib.sha256(f"F2WP1207_ENTITY/v1:{OWNER}:{db_path}".encode()).hexdigest()[:32],
        "installation_id": inst,
        "state_root_id": hashlib.sha256(f"F2WP1207_STATE_ROOT/v1:{inst}:{db_path}".encode()).hexdigest()[:32],
        "binding_id": hashlib.sha256(f"F2WP1207_HOST_BINDING/v1:{inst}:local".encode()).hexdigest()[:32],
    }


def genesis_identity(conn: sqlite3.Connection, db_path: str, star_dir: str) -> dict:

    ADOPT FIRST (2026-10-15): A already has the real donor identity layer (entity 2026-09-03, >800 epochs, other column layout than schema_f2.sql). If an epoch chain exists,
    continue the ids of its LATEST epoch and write NO identity rows (the positional INSERTs below failed 'has 3 columns but 5 values' since 2026-10-07)."""
    try:
        r = conn.execute("SELECT e.state_root_id,e.installation_id,e.host_binding_id,i.entity_id FROM f2_runtime_epoch e "
                         "JOIN f2_installation_identity i ON i.installation_id=e.installation_id ORDER BY e.started_at DESC LIMIT 1").fetchone()

def epoch_for_session(conn: sqlite3.Connection, ids: dict, session_id: str) -> str:
    """One runtime epoch per Claude session; predecessor = most recent epoch."""
    epoch_id = hashlib.sha256(f"F2WP1207_EPOCH/v1:{ids['installation_id']}:{session_id}".encode()).hexdigest()[:32]
    if conn.execute("SELECT 1 FROM f2_runtime_epoch WHERE runtime_epoch_id=?", (epoch_id,)).fetchone():
        return epoch_id
    prev = conn.execute(
        "SELECT runtime_epoch_id FROM f2_runtime_epoch WHERE installation_id=? ORDER BY started_at DESC, rowid DESC LIMIT 1",
        (ids["installation_id"],)).fetchone()
    conn.execute(
        "INSERT INTO f2_runtime_epoch (runtime_epoch_id,state_root_id,installation_id,host_binding_id,"
        "started_at,predecessor_epoch_id,schema,session_id) VALUES (?,?,?,?,?,?,?,?)",
        (epoch_id, ids["state_root_id"], ids["installation_id"], ids["binding_id"], _now(),
         prev[0] if prev else None, SCHEMA, session_id))
    conn.commit()
    return epoch_id


def _turn_sync(conn: sqlite3.Connection, db_path: str, star_dir: str, session_id: str,
         prompt: str, retrieval: str | None) -> dict | None:
    """Run one canonical turn. Returns a small dict, or None if gate off.
    Raises on real failure (caller logs). conn must be a plain sqlite3 connection."""
    if not tl.turn_loop_active(conn):
        return None
    ensure_f2_schema(conn)
    ids = genesis_identity(conn, db_path, star_dir)
    epoch = epoch_for_session(conn, ids, session_id)
    conn.row_factory = sqlite3.Row
    kw = {}
    if _o08b_q is not None:  # 0.8B-Organ als Quelle fuer modell_* + abstrakt_* (laeuft im abgekoppelten Prozess, kein Hook-Limit; fehlt die Messung -> Achsen None)
        try:
            kw = _o08b_q.felder(prompt, lane="hook", session=session_id, db_path=db_path, timeout_s=5.0).get("kwargs") or {}
        except Exception:
            kw = {}
    entry = tl.TypedEntry(
        session_id=session_id, runtime_epoch_id=epoch, event_type="UserPromptSubmit",
        prompt_len=len(prompt or ""), retrieval_present=bool(retrieval),
        retrieval_len=len(retrieval or ""), **kw)
    res = tl.run_canonical_turn(conn, entry)
    if res is None:
        return None
    try:  # FRANK3 stage E: GRID10 v2 SHADOW lane (computes + logs next to the legacy winner; never drives, never injects)
        _v2_shadow(conn, session_id, prompt, retrieval, res.jspace.winner_cell_id)
    except Exception:
        pass
    try:  # extended-signal x-lane: own tables, read by nothing, publishes nothing
        _v2x_shadow(conn, session_id, prompt, retrieval, res.jspace.winner_cell_id)
    except Exception:
        pass
    return {
        "frame_id": res.frame_id, "reentry": res.reentry.kind,
        "winner": res.jspace.winner_cell_id,
        "broadcast": getattr(res.jspace, "broadcast_value", None),
        "epoch": epoch,
    }

def turn(conn, db_path, star_dir, session_id, prompt, retrieval):
    """ASYNC wrapper (2026-10-15). stern.py's UserPromptSubmit hook has a HARD 3 s budget (138 'time limit reached' in 8 h), the canonical turn writes ~15 rows and can
    wait on the busy DB. So: gate check on the given conn, start the real turn DETACHED (own process, 8 s busy timeout) and return the spooled result of the
    PREVIOUS prompt of this session (same dict shape as _turn_sync; None if none yet). One run per session in flight."""
    import json, os, subprocess
    if not tl.turn_loop_active(conn):
        return None
    base = os.path.join(os.path.expanduser(os.environ.get("FRANK3_HIST_DIR") or "~/.local/share/frank3/hist"), "grid10")
    sid = "".join(ch for ch in str(session_id) if ch.isalnum() or ch == "-")[:60] or "x"
    spool, lock = os.path.join(base, "turn_%s.json" % sid), os.path.join(base, "run_%s.lock" % sid)
    prev = None
    try:
        if os.path.exists(spool):
            prev = json.load(open(spool))
            os.replace(spool, spool + ".done")
    except Exception:
        prev = None
    try:
        os.makedirs(base, exist_ok=True)
        if not (os.path.exists(lock) and time.time() - os.path.getmtime(lock) < 20):
            open(lock, "w").close()
            pr = subprocess.Popen([sys.executable, "-I", "-c", _ASYNC_SNIPPET, str(_HERE), str(db_path), str(star_dir), str(session_id), spool, lock], stdin=subprocess.PIPE, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, start_new_session=True)
            pr.stdin.write(((prompt or "")[:4000] + chr(0) + (retrieval or "")[:20000]).encode("utf-8", "replace"))
            pr.stdin.close()
    except Exception:
        pass
    return prev

def _appraise(text):
    """Lexicon appraisal of the prompt via the production kogloop.appraise_ext (modules/hist/kogloop.py:148: German/English phrase lexicons, negation-aware,
    pure function). HEURISTIC (lexical), not a model measurement. Loaded lazily by path with sys.path restored (kogloop inserts its dir into sys.path).
    Any failure -> None -> the cells stay 0."""
    global _APPRAISE
    if _APPRAISE is None:
        try:
            import importlib.util as _iu
            saved = list(sys.path)
            try:
                sp = _iu.spec_from_file_location("grid10_kogloop_lex", str(_HERE.parent / "hist" / "kogloop.py"))
                m = _iu.module_from_spec(sp)
                sp.loader.exec_module(m)
                _APPRAISE = m.appraise_ext
            finally:
                sys.path[:] = saved
        except Exception:
            _APPRAISE = False
    if not _APPRAISE:
        return None
    try:
        return _APPRAISE(text)
    except Exception:
        return None


def v2_signals(conn, prompt, retrieval, extended=True, hist=None, session=None):
    """Frank's MEASURABLE grid inputs, mirroring OMEGA's honest wiring (omega_grid_signal.go DeriveGridSignalVector).
    LEGACY (extended=False, what the main lane f3_grid_v2_* still uses -- see _v2_shadow): G10, G6, G7 only.
    EXTENDED (extended=True, fed ONLY to the separate x-lane f3_grid_v2x_*):
      G10 Umfang = min(1, est. prompt tokens/3000)                       (measured length; est. = chars/4)
      G6 Werkzeugdruck = 1 iff prompt looks like a code/tool task        (regex HEURISTIC)
      G7 Zielkontinuitaet = a goal is active (architect mode or open omega_goal)   (state lookup)
      G2 Affekt-Intensitaet = max(frustration, praise, urgency) lexicon weight     (HEURISTIC, kogloop.appraise_ext; OMEGA: 4B percept axis Intensitaet)
      G3 Valenz-Spannung = |praise - max(frustration, correction)|                  (HEURISTIC, same lexicon; OMEGA: |4B valence|)
      G8 Gedaechtnisdruck = min(1, n_items/5), n_items = bullet lines ("  • ") of the retrieval block the hook already fetched for this prompt
         (PROXY: OMEGA sums Salience of TITAN candidates /5; the hook passes only text, no per-item salience, so each item counts 1 = upper bound)
      G1,G4,G5,G9 (2026-11-06): lexical_fallback port of OMEGA omega_lexical_percept.go (modules/grid10/lexical_percept.py; surface features, calibrated on nothing).
         G4 = (b+1)/2 >= 0.5 for any non-empty prompt (Go semantics). G9 = 1 - max Jaccard vs the last 8 user msgs of this SESSION (hist: list of (digest, token-hashes),
         default: read f3_grid_v2x_hist for `session`; no history -> G9 stays 0, never invented). The x-lane only; writing the ring is _v2x_shadow's job (this fn never writes).
    OLD NOTE (before 2026-11-06; G1,G4,G5,G9 are now the lexical fallback above, G2/G3 unchanged from kogloop): G1 Abstraktion, G4 Bindung, G5 Endgueltigkeit (OMEGA: 4B percept axes; the 4B/probe is not run
      in the hook, abstrakt_* TypedEntry fields are off (ABSTRAKTION_ANBINDUNG_AKTIV=0); f2_grid10_percept.signale_json G1..G5 are NOT these axes:
      they are per-cell random projections of the world vector, 'function-neutral' by design), G9 Ueberraschung (needs model logprobs; hook lane has
      modell_ueberraschung=0 in every row). No hash/random values anywhere. Fail-silent per cell."""
    import re
    global _CODE_CUE
    if _CODE_CUE is None:
        _CODE_CUE = re.compile(r"```|\b(def|class|import|git|pip|bash|python|sqlite|systemctl|traceback|error|fix|bug|build|install|script|datei|file|run)\b|[/~][\w.-]+/[\w.-]+", re.I)
    p = prompt or ""
    sig = {"G%d" % i: 0.0 for i in range(1, 11)}
    sig["G10"] = min(1.0, (len(p) / 4.0) / 3000.0)
    sig["G6"] = 1.0 if _CODE_CUE.search(p) else 0.0
    g7 = 0.0
    try:
        r = conn.execute("select wert from star_konfig where schluessel='architekt.aktiv'").fetchone()
        if r and str(r[0]) == "1":
            g7 = 1.0
        elif conn.execute("select 1 from omega_goal where lower(status) not in ('done','closed','abandoned','complete','completed') limit 1").fetchone():
            g7 = 1.0
    except Exception:
        pass
    sig["G7"] = g7
    if not extended:
        return sig
    try:
        ap = _appraise(p)
        if ap:
            fr, pr, ur, co = (float(ap.get(k, 0.0) or 0.0) for k in ("frustration", "praise", "urgency", "correction"))
            sig["G2"] = max(0.0, min(1.0, max(fr, pr, ur)))
            sig["G3"] = max(0.0, min(1.0, abs(pr - max(fr, co))))
    except Exception:
        pass
    try:  # lexical_fallback G1,G4,G5,G9 (G2/G3 deliberately NOT taken from it: they stay kogloop)
        if p.strip():
            import lexical_percept as _lp
            if hist is None:
                hist = []
                try:
                    import json as _j
                    hist = [(r[0], _j.loads(r[1])) for r in reversed(conn.execute(
                        "select digest,toks from f3_grid_v2x_hist where session=? order by id desc limit 8", (session or "",)).fetchall())]
                except Exception:
                    hist = []
            ax = _lp.lexical_axes(p[:30000], hist)  # cap: 30k chars ~ 100 ms; hook budget 3 s
            for k in ("G1", "G4", "G5"):
                sig[k] = float(ax[k])
            if ax["g9_valid"]:
                sig["G9"] = float(ax["G9"])
    except Exception:
        pass
    try:
        if retrieval:
            n = sum(1 for ln in str(retrieval).splitlines() if ln.startswith("  \u2022 "))
            sig["G8"] = min(1.0, n / 5.0)
    except Exception:
        pass
    return sig

_ready = set()


def connect(db=None, timeout=0.8):
    c = sqlite3.connect(db or DB, timeout=timeout)
    c.execute("pragma busy_timeout=%d" % int(timeout * 1000))
    if (db or DB) not in _ready:
        c.execute("pragma journal_mode=wal")
        for d in DDL:
            c.execute(d)
        c.commit()
        _ready.add(db or DB)
    return c


def _clamp(x, lo=0.0, hi=1.0):
    try:
        return lo if x < lo else hi if x > hi else float(x)
    except Exception:
        return lo


def prefix(src):
    s = (src or "").split(":", 1)[0] or "(none)"
    return s[:24]


def _count(c, name, n=1):
    c.execute("insert into f3_gwt_counter(name,v) values(?,?) on conflict(name) do update set v=v+?", (name, n, n))


def _note(c, route, cands):
    for x in cands:
        c.execute("insert into f3_gwt_ledger(route,prefix,n) values(?,?,1) on conflict(route,prefix) do update set n=n+1", (route, prefix(x["source"])))


def mode(c):
    try:
        if (c.execute("select wert from star_konfig where schluessel='frank3.kill'").fetchone() or ("0",))[0] == "1":
            return "off"
        return (c.execute("select wert from star_konfig where schluessel='frank3.gwt.mode'").fetchone() or ("shadow",))[0]
    except Exception:
        return "off"


_PIX = re.compile(r"data:image/|^\x89PNG|^GIF8|^\xff\xd8\xff")
_SECRET = re.compile(r"(?i)(sk-[A-Za-z0-9_\-]{12,}|gsk_\S+|nvapi-\S+|Bearer\s+\S+|(api[_-]?key|token|secret|passw\w*)\s*[=:]\s*\S+)")


def _budget(c, n):
    day = time.strftime("%Y-%m-%d", time.gmtime())
    c.execute("insert or ignore into omega_dbw_day(day,bytes,writes,denied) values(?,0,0,0)", (day,))
    used = c.execute("select bytes from omega_dbw_day where day=?", (day,)).fetchone()[0]
    if used + n > DAY_CAP:
        c.execute("update omega_dbw_day set denied=denied+1 where day=?", (day,))
        return False
    c.execute("update omega_dbw_day set bytes=bytes+?, writes=writes+1 where day=?", (n, day))
    return True


# ---------------------------------------------------------------------------------------------- THE publish site
def publish(producer, cand, cost_tokens=0, conn=None):
    """THE ONLY publish site (static scanner enforces). Never raises, never blocks long. Returns False when not accepted (counted as missed)."""
    own = conn is None
    c = conn
    try:
        c = c or connect()
        if producer not in PRODUCERS or not isinstance(cand, dict) or mode(c) == "off":
            _count(c, "published_missed")
            c.commit()
            return False
        pay = str(cand.get("payload", ""))
        if _PIX.search(pay):
            _count(c, "published_missed")
            c.commit()
            return False
        pay = _SECRET.sub("<redacted>", pay)[:PAYLOAD_MAX]
        row = (time.time(), producer, str(cand.get("id") or "%s:%x" % (producer, hash(pay) & 0xffffffff))[:80], str(cand.get("kind", "event"))[:40],
               str(cand.get("source", producer))[:80], str(cand.get("goal_ref", ""))[:60], pay, _clamp(cand.get("salience", 0.3)),
               _clamp(cand.get("confidence", 0.8)), _clamp(cand.get("urgency", 0.0)), str(cand.get("origin") or "body")[:20],
               str(cand.get("trust") or "measured")[:20], int(cost_tokens))
        if not _budget(c, len(pay) + 120):
            _count(c, "published_missed")
            c.commit()
            return False
        c.execute("insert into f3_gwt_pending(ts,producer,cid,kind,source,goal_ref,payload,salience,confidence,urgency,origin,trust,cost) values(?,?,?,?,?,?,?,?,?,?,?,?,?)", row)
        over = c.execute("select count(*) from f3_gwt_pending").fetchone()[0] - PENDING_MAX
        if over > 0:
            c.execute("delete from f3_gwt_pending where seq in (select seq from f3_gwt_pending order by seq limit ?)", (over,))
            _count(c, "pending_dropped", over)
        _count(c, "published_ok")
        _note(c, "publish", [{"source": row[4]}])
        c.commit()
        return True
    except Exception:
        try:
            c.rollback()
            _count(c, "published_missed")
            c.commit()
        except Exception:
            pass
        return False
    finally:
        if own and c is not None:
            try:
                c.close()
            except Exception:
                pass


def _row_to_cand(r, now):
    ts, producer, cid, kind, source, goal_ref, payload, sal, conf, urg, origin, trust, cost = r
    age = max(0.0, now - ts)
    return {"id": cid, "kind": kind, "source": source, "goal_ref": goal_ref, "payload": payload, "salience": sal * math.pow(0.5, age / HALF_LIFE_S),
            "confidence": conf, "urgency": urg, "origin": origin, "trust": trust, "producer": producer, "cost": cost, "ts": ts, "age": age}


def drain(conn, now=None):
    """returns pending candidates re-scored for staleness (on arrival, like OMEGA) and empties the queue."""
    now = now or time.time()
    rows = conn.execute("select ts,producer,cid,kind,source,goal_ref,payload,salience,confidence,urgency,origin,trust,cost from f3_gwt_pending order by seq").fetchall()
    conn.execute("delete from f3_gwt_pending")
    _count(conn, "drained", len(rows))
    return [_row_to_cand(r, now) for r in rows]

# ---------------------------------------------------------------------------------------------- competition
def _cells(conn):
    try:
        r = conn.execute("select j from f3_grid_v2_state where singleton=1").fetchone()
        return json.loads(r[0])["cells"] if r else {}
    except Exception:
        return {}


def _boost(conn, kind):
    r = conn.execute("select a,b from f3_gwt_boost where kind=?", (kind,)).fetchone()
    return (r[0] / (r[0] + r[1])) if r else 0.5


def compete(conn, cands, session="", budget_tokens=60, max_n=2, min_score=0.2):
    """score = salience * (1+0.5*urgency) * grid gain (v2 cell of the kind's cell, clamped 0..2) * novelty (recent broadcasts) * learned boost.
    Winners must beat min_score and fit the token budget. Abstain = empty list."""
    cells = _cells(conn)
    recent = {}
    for (cids,) in conn.execute("select cids from f3_gwt_broadcast order by id desc limit 20").fetchall():
        for x in cids.split(","):
            recent[x] = recent.get(x, 0) + 1
    scored = []
    for x in cands:
        gain = 1.0 + 0.25 * _clamp(cells.get(KIND_CELL.get(x["kind"], ""), 0.0), 0.0, 2.0)
        s = x["salience"] * (1 + 0.5 * x["urgency"]) * gain / (1 + recent.get(x["id"], 0)) * (0.75 + 0.5 * _boost(conn, x["kind"]))
        scored.append((s, x))
    scored.sort(key=lambda t: -t[0])
    win, used, seen = [], 0, set()
    for s, x in scored:
        t = max(1, len(x["payload"]) // 4)
        if s >= min_score and x["id"] not in seen and used + t <= budget_tokens and len(win) < max_n:
            win.append(x)
            used += t
            seen.add(x["id"])
    return win


def format_line(win):
    if not win:
        return ""
    return ("GWT: " + " | ".join("[%s] %s" % (x["kind"], x["payload"][:110]) for x in win))[:260]


def cycle(session="", prompt="", db=None, now=None):
    """one workspace cycle (called by the UserPromptSubmit hook): drain pushes + pull extras -> compete -> broadcast (log) -> line.
    Returns (line_to_inject_or_'', info). shadow: line='' but everything is logged with injected=0."""
    t0 = time.perf_counter()
    c = connect(db)
    try:
        md = mode(c)
        if md == "off":
            return "", {"mode": "off"}
        try:  # mathlab: exact-calc candidate through THE publish site (producer organ); bus mode decides injection
            import sys as _s
            _o = os.path.join(os.path.dirname(os.path.abspath(__file__)), "modules", "organs")
            if _o not in _s.path:
                _s.path.insert(0, _o)
            import gwt_glue
            _cc = gwt_glue.calc_candidate(c, prompt)
            if _cc:
                publish("organ", _cc, conn=c)
        except Exception:
            pass
        try:  # kognition (historic organs via kogloop): shadow = measured + counted in f3_run, nothing published; active publishes one structured candidate
            import gwt_glue as _gg
            _kc = _gg.kog_candidate(c, prompt, session)
            if _kc:
                publish("organ", _kc, conn=c)
        except Exception:
            pass
        cands = drain(c, now)
        for cs in extra_sources(c, now).values():
            cands.extend(cs)
        win = compete(c, cands, session)
        line = format_line(win)
        tokens = (len(line) + 3) // 4 if line else 0
        inj = 1 if (md == "active" and line) else 0
        us = int((time.perf_counter() - t0) * 1e6)
        c.execute("insert into f3_gwt_broadcast(ts,session,cids,kinds,line,tokens,injected,n_cand,us) values(?,?,?,?,?,?,?,?,?)",
                  (time.time(), session[:40], ",".join(x["id"] for x in win), ",".join(x["kind"] for x in win), line, tokens, inj, len(cands), us))
        c.execute("delete from f3_gwt_broadcast where id < (select max(id) from f3_gwt_broadcast) - 500")  # bounded retention
        _count(c, "broadcasts")
        c.commit()
        return (line if inj else ""), {"mode": md, "n_cand": len(cands), "winners": [x["id"] for x in win], "tokens": tokens, "us": us}
    except Exception as e:
        return "", {"error": str(e)[:80]}
    finally:
        c.close()


# ---------------------------------------------------------------------------------------------- credit (soft, per kind)
def credit(conn, kind, value):
    """value in [0,1] fractional reward (omega_gwt_credit idea: credit in {0,1} == hard reward). Beta(a,b) per candidate kind -> boost."""
    v = _clamp(value)
    conn.execute("insert into f3_gwt_boost(kind,a,b) values(?,1+?,1+?) on conflict(kind) do update set a=a+?, b=b+?", (kind, v, 1 - v, v, 1 - v))


# ---------------------------------------------------------------------------------------------- law: ledger, conservation, scanner support
def ledger_snapshot(conn):
    routes = {}
    for r, p, n in conn.execute("select route,prefix,n from f3_gwt_ledger").fetchall():
        routes.setdefault(r, {})[p] = n
    counters = dict(conn.execute("select name,v from f3_gwt_counter").fetchall())
    return {"routes": routes, "counters": counters, "pending": conn.execute("select count(*) from f3_gwt_pending").fetchone()[0]}


def unregistered_routes(conn):
    return sorted(r for (r,) in conn.execute("select distinct route from f3_gwt_ledger").fetchall() if r not in ROUTES)


def conservation(conn):
    """LAW: every accepted publish is exactly one of {still pending, drained, dropped}; ledger['publish'] == published_ok; no unregistered route."""
    s = ledger_snapshot(conn)
    k = s["counters"]
    ok_pub = k.get("published_ok", 0)
    led_pub = sum(s["routes"].get("publish", {}).values())
    accounted = k.get("drained", 0) + k.get("pending_dropped", 0) + s["pending"]
    return {"published_ok": ok_pub, "ledger_publish": led_pub, "accounted": accounted, "missed": k.get("published_missed", 0),
            "pending_dropped": k.get("pending_dropped", 0), "holds": ok_pub == led_pub == accounted and not unregistered_routes(conn),
            "unregistered": unregistered_routes(conn)}


def export_jsonl(winners_or_cands, path=OUTBOX):
    """wire format v=1 for the OMEGA side (JSON lines, fields WIRE_FIELDS). Append-only, bounded by caller."""
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "a", encoding="utf-8") as f:
        for x in winners_or_cands:
            d = {k: x.get(k) for k in WIRE_FIELDS if k != "v"}
            d["v"] = 1
            f.write(json.dumps(d, ensure_ascii=False) + "\n")
